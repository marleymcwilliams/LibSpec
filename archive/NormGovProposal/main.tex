\section{Continuous Q* Thread (Data \textrightarrow{} Reality \textrightarrow{} Normative)}
We maintain three \emph{coupled} objects at time $t$:
\begin{itemize}
  \item $Q^\*_{\mathrm{hist}}(t)$: history-consistent posterior from data,
  \item $Q^\*_{\mathrm{now}}(t)$: compressed, decision-usable reality,
  \item $Q^\*_{\mathrm{norm}}(t)$: feasible normative target under constraints.
\end{itemize}
They are updated by a single operator $\mathbf{T}_{\mathrm{thread}}$:
\begin{align*}
Q^\*_{\mathrm{hist}}(t) &= \operatorname{Post}(\mathcal{D}_t;\,\text{priors, model}),\\
Q^\*_{\mathrm{now}}(t)  &= \operatorname{Compress}(Q^\*_{\mathrm{hist}}(t)),\\
Q^\*_{\mathrm{norm}}(t) &= \operatorname{Close}(Q^\*_{\mathrm{now}}(t),\,\Phi).
\end{align*}

\paragraph{Thread update $\mathbf{T}_{\mathrm{thread}}$.}
\emph{Assimilate} new data $\to Q^\*_{\mathrm{hist}}$; 
\emph{Reconcile} to $Q^\*_{\mathrm{now}}$; 
\emph{Plan} via local evaluation around $Q^\*_{\mathrm{now}}$ (chance, co-players, terminal conditions); 
\emph{Close} to $Q^\*_{\mathrm{norm}}$ under constraints; 
\emph{Guard} invariants; 
\emph{Act} with the smallest admissible step; 
\emph{Prove} with a receipt and \emph{Integrate} into snapshots.

\paragraph{Invariants.}
(i) Data-consistency: $\|Q^\*_{\mathrm{now}}-Q^\*_{\mathrm{hist}}\|\leq \epsilon_t$. 
(ii) Feasible intent: $Q^\*_{\mathrm{norm}}$ satisfies resource and rule constraints. 
(iii) Monotone receipts: applied actions reduce $\mathrm{dist}(Q^\*_{\mathrm{now}}, Q^\*_{\mathrm{norm}})$.

\paragraph{Speed governance.}
Step size and planning depth adapt to \emph{Clarity} (intent legibility), \emph{Certainty} (posterior support), and \emph{Tractability} (constraint satisfaction). Low scores \emph{slow} the step (shorter moves, deeper evaluation); high scores allow \emph{faster} integration.

\section{Local Evaluator (Coalition Shuffling)}
We use a state-space evaluator with decision, chance, and terminal nodes. Chance nodes aggregate
by expectation; decision nodes optimize the active coalition's utility component; terminal nodes
close with value vectors. Practical uses:
\begin{enumerate}
  \item \textbf{Counterfactuals}: test whether the proposed move improves our position under modeled
  rational/cooperative play.
  \item \textbf{Local policies}: extract minimal admissible actions that move $Q^\*_{\mathrm{now}} \to Q^\*_{\mathrm{norm}}$.
  \item \textbf{Stress}: perturb chance weights/opponent models to assess robustness pre-commit.
\end{enumerate}
This evaluator can be called at any point on the strand; its transcript is included in the \textit{receipt} for reproducibility.

\section{Workflow: Recipes \textrightarrow{} Contracts \textrightarrow{} Gates}
\subsection{Compose with ResearchHub}
Recipes: alignment, agentic-guided execution, iterative change. Inject scope labels, pre-commit
output, and optional repo ZIP. Outputs enumerate SPEC anchors to update and explicit must-fail
reason codes.

\subsection{Map to Layer Contracts (Quick Check)}
\begin{itemize}
  \item \textbf{Rep/Alg}: stratified/Filippov setting; budget frontier dwell $\tau_{\min}$ and defect $C_{\mathrm{fr}}$.
  \item \textbf{Stoch/Inf}: finite dictionaries within EF radius; whitened cumulants.
  \item \textbf{Clos}: DRO soft-transport; KKT stationarity; adaptive radius policy.
  \item \textbf{Ctrl}: Dirac projections; PR checks; ISS margins and freeze windows.
  \item \textbf{Comp}: non-expansive functors; Lipschitz constants in FIM metric.
\end{itemize}

\subsection{Land via WorkCards (Guards \& Evidence)}
Declare precise \textit{Admits/Emits/Guards/FailFast/Links}; evaluate guards over canonical inputs \& policy digests.
Each execution emits a \code{ReceiptEnvelope/v1} bound by \code{checker_hash} $\in$ EnvLock.

\subsection{Phase Gates (Now \textrightarrow{} Next)}
\begin{itemize}
  \item \textbf{PR-D} Saga/Compensation: side-effecting tools declare compensation; negative \code{tool_missing_compensation}.
  \item \textbf{PR-E} LLM I/O Envelope: typed envelope, rounding receipts; negatives \code{llm_missing_receipt}, \code{llm_untyped_io}.
  \item \textbf{PR-F/G/H}: SSOT tooling; UpdateCert generator; formal ResearchLoop with ``ideas-as-prose'' negative.
\end{itemize}

\section{Checklists}
\subsection{Proposal (Spark)}
\begin{itemize}
  \item Recipe rendered (attach output)
  \item SPEC anchor(s) to update listed (path/digest)
  \item New negative(s) + reason\_code(s) enumerated
\end{itemize}
\subsection{Alignment (Stabilize)}
\begin{itemize}
  \item Living Doc snapshot updated
  \item Master Plan acceptance bars linked
  \item SSOT pointers verified (no prose duplication)
\end{itemize}
\subsection{Governance (Harden)}
\begin{itemize}
  \item WorkCard fields complete
  \item Governance class \& approval\_chain wired; policy\_version pinned
  \item Negative(s) created and added to coverage
\end{itemize}
\subsection{Proof (Invert)}
\begin{itemize}
  \item ReceiptEnvelope/v1 emitted; \code{checker_hash} $\in$ EnvLock
  \item Negatives fail for intended reasons (Gate-0 \& CI)
  \item Replay passes under declared portability stance
\end{itemize}
\subsection{Land (Integrate)}
\begin{itemize}
  \item Small PR only; diffs reference SPEC anchors/tests
  \item Phase gate progress recorded in Master Plan
\end{itemize}
