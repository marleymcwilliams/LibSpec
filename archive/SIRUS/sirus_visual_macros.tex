\newcommand{\SirusBox}[4]{%
  \par\smallskip\noindent
  \begingroup\setlength{\fboxsep}{\sirusPad}%
  \fcolorbox{#1}{#2}{%
    \begin{minipage}{\sirusBoxWidth}\small
      {\bfseries #3}\par\medskip
      #4
    \end{minipage}}%
  \endgroup\par\smallskip}
