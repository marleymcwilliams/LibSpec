\section{Failure Cookbook \& Operations Runbook (v2)}\label{sec:runbook-v2}

\paragraph{Intent.}
Codify \emph{deterministic}, proof-carrying responses to certificate failures. For each trigger we list (i) \textbf{Detection}, (ii) \textbf{Immediate action} (always safe and reversible), (iii) \textbf{Stabilization} (what must hold before unfreezing), and (iv) \textbf{Evidence} (Tier-1/2 fields and IDs). All actions must preserve the guard order printed in the window’s \emph{proof print} (EF$\rightarrow$DRO$\rightarrow$PR$\rightarrow$CBF$\rightarrow$\omega$, Subsec.~\ref{subsec:det-replay}).

\subsection{Freeze windows, dwell management, and tube hygiene}\label{subsec:freeze-hygiene}
\emph{\layertag{Geom/Ctrl/Rep}}

\begin{definition}[Freeze window]\label{def:freeze-window}
A contiguous index set $W$ where Rule~\ref{rule:frontier-freeze} applies. During $W$, updates to actuation basis, $\mathcal U$ policy, and CIS are disabled; CLF/CBF-QP uses the last off-tube basis and tube-aware audit (Lemma~\ref{lem:cbf-salt}); KPI rows carry \texttt{proof\_print\_id} with PR witness present but not \emph{consumed} for control updates until unfreeze.
\uses{Rule~\ref{rule:frontier-freeze}; Lem.~\ref{lem:cbf-salt}; Subsec.~\ref{subsec:det-replay}.}
\end{definition}

\begin{proposition}[Dwell \& tube retune]\label{prop:tube-retune}
If grazing probability $\constref{pgraze}$ exceeds $\constref{pthr}$ or $\mathrm{CI}(\constref{nuhat})$ touches $0$, set
$\constref{taumin}\leftarrow (1+\constref{gammadwell})\constref{taumin}$ and
$\constref{rtube}\leftarrow \constref{rtube}/(1+\constref{gammadwell})$, and log the change with tube-id and timestamps. Revert only after two windows with positive normal-speed CI and no chatter events.
\uses{Rule~\ref{rule:grazing}; (C1) $\constref{rtube}$; (C3) $\constref{taumin}$; (C8) $\constref{pgraze},\constref{gammadwell},\constref{nuhat}$.}
\end{proposition}

\subsection{Runbook checks: what operators must verify}\label{subsec:runbook-checks}
\emph{\layertag{Rep}}

\begin{acceptchecks}
\item \textbf{Trigger fidelity:} for each row in Tab.~\ref{tab:fail-cookbook}, the log shows the matching trigger value, the exact timestamped immediate action, and the unfreeze event that met the listed criteria; KPI rows and the Ladder reflect the same sequence.
\item \textbf{Freeze integrity:} during any Freeze window (Def.~\ref{def:freeze-window}), CIS is unchanged; solver transcripts reference the frozen basis; PR witnesses produced in $W$ are recorded but \emph{not} consumed by control until unfreeze; \texttt{proof\_print\_id} is present.
\item \textbf{Dwell/tube hygiene:} if dwell escalated, evidence shows updated $\constref{taumin}$ and $\constref{rtube}$, plus positive normal-speed CI before unfreeze (Prop.~\ref{prop:tube-retune}).
\item \textbf{Rights-plane safety:} any window with $\mathrm{LB}<0$ leads to $\omega$-run rejection (Lemma~\ref{lem:cbf-salt}); CVaR breaches show Rule~\ref{rule:rights-cvar} transcript and freeze timestamps.
\item \textbf{Quarantine propagation:} when \texttt{quarantine\_bit}=1, there are no new CIS rows; automata reject any attempt to consume quarantined rows (Subsec.~\ref{subsec:ladder-prop}).
\item \textbf{Clamp/FDR discipline:} $(r,\cutq)$ shrink/relax events match clamp transitions and realized-FDR rows; promotions frozen outside Green (Rule~\ref{rule:ef-clamp}).
\item \textbf{Budget guard compliance:} timeouts flip to conservative mode and clear before promotions resume; \texttt{budget\_guard\_verdict} appears in Tier-1 and KPI rows (Def.~\ref{def:budget}, Prop.~\ref{prop:slo}).
\end{acceptchecks}
