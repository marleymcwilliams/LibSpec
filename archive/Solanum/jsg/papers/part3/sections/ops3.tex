% ==== Part III — Operational Contracts and Evidence (v2) =====================
\section{Operational Contracts and Evidence (v2)}\label{sec:ops-v2}

\paragraph{Scope.}
This section makes the I\textsubscript{1}–I\textsubscript{5} guarantees operational by adding grazing-aware dwell control, an EF guard with a leakage clamp, promotion gates that privilege identifiability over correlation, frontier-aware CBF relaxation, PR/GKYP witnesses, switching-stability policies, and proof-carrying telemetry/provenance packs. New or clarified pieces are marked (Δ). A \emph{window proof print} (Δ) is emitted each window to record the fixed guard-consumption order: EF$\rightarrow$DRO$\rightarrow$PR$\rightarrow$CBF$\rightarrow\omega$.

\subsection{(a) Geometric diagnosis: DEC/Hodge, holonomy, dwell, Fisher continuity}\label{subsec:ops-geom}
\emph{\layertag{Rep/Alg/Inf/Ctrl}}

\begin{rulex}{Frontier projection freeze (Δ)}\label{rule:frontier-freeze}
When inside any frontier tube of radius $\constref{rtube}$, \emph{freeze} basis updates and project control to the interior half-space aligned with the certified normal. Concretely: (i) hold the actuation plane/basis fixed; (ii) zero all promotions; (iii) project $u$ by $u\leftarrow \Pi_{\mathrm{safe}}(u)$ using the previous off-tube basis; (iv) log \texttt{freeze\_start/end} and the tube id; (v) include \texttt{cbf\_audit\_id} in the window proof print. Unfreeze only after a completed dwell interval $\ge\constref{taumin}$ with positive normal speed CI.
\uses{(C1) $\constref{rtube}$, (C3) $\constref{taumin}$.}
\end{rulex}

\begin{rulex}{Grazing-aware dwell escalation (Δ)}\label{rule:grazing}
Maintain a bootstrap CI for the frontier-normal speed and a graze probability $\constref{pgraze}$ with threshold $\constref{pthr}$. If $\mathrm{CI}(\constref{nuhat})$ hits $0$ or $\constref{pgraze}>\constref{pthr}$, then increase dwell and narrow tubes:
\[
\constref{taumin}\leftarrow (1+\constref{gammadwell})\,\constref{taumin},\qquad
\constref{rtube}\leftarrow \tfrac{1}{1+\constref{gammadwell}}\,\constref{rtube}.
\]
Apply the Frontier projection freeze during the escalated dwell (Rule~\ref{rule:frontier-freeze}).
\uses{(C1) $\constref{rtube}$; (C3) $\constref{taumin}$; (C8) $\constref{nuhat},\constref{pgraze},\constref{gammadwell}$.}
\end{rulex}

\begin{lemma}[Frontier spend ledger and freeze-aware shape bias]\label{lem:ledger-freeze}
\emph{[\textup{Alg/Stoch}]}
Track the Jacobi defect as an additive \emph{Frontier Spend Ledger} with per-crossing charge $C_{\mathrm{fr}}$ and No-Zeno count $N_{\mathrm{cross}}$. Under the freeze rule, shape-derivative bias over a window of length $\tau$ obeys
\[
\mathrm{Bias}=O\!\Big(\frac{T_{\mathrm{frz}}}{\tau}\Big),\qquad
\int_0^T\!\|\Delta_{\mathrm{fr}}\|\le C_{\mathrm{fr}}\,N_{\mathrm{cross}}(0,T).
\]
\uses{\jsgiLem{\lblJacobiDefect}; \jsgiLem{\lblNoZeno}; (C3) $\constref{taumin}$.}
\end{lemma}

\begin{definition}[Holonomy and harmonic rotational mass (Δ)]\label{def:hol-harm}
Let $\vartheta_j$ be principal-angle increments across crossings and $\delta_{\mathrm{gap},j}$ the spectral gap. Define
\[
\constref{Hol}(0,T):=\sum_j \vartheta_j,\qquad
\constref{Harm}:=\sum_j \frac{\vartheta_j}{\max(\delta_{\mathrm{gap},j},\eta_{\mathrm{cal}})}.
\]
If \constref{Harm} rises persistently, \emph{k-escalate} the actuation plane before allowing basis updates.
\uses{(C9) $\constref{Hol},\constref{Harm},\eta_{\mathrm{cal}}$; \jsgiiSec{\lblSchemas}\,(\textit{DK/Wedin calibration record}).}
\end{definition}

\begin{rulex}{EF guard and leakage clamp FSM (Δ)}\label{rule:ef-clamp}
Compute $r=\varepsilon/L_\psi$ and enforce \texttt{kl\_quad\_bounds()} before using Fisher-quadratic proxies. Generate a unique \texttt{ef\_guard\_id} for the verdict and place it in the window proof print. Run a clamp FSM on leakage $\chi(t)$:
Green ($\chi\!\le\!\constref{chiG}$): normal;\quad
Yellow ($\constref{chiG}\!<\!\chi\!\le\!\constref{chiR}$): shrink $(r,q)$, freeze promotions;\quad
Red ($\chi\!>\!\constref{chiR}$): fail-closed to nonparametrics (HSIC, $\mathrm{CVaR}$ tails) until $\chi<\constref{chiY}$.
\uses{\jsgiiThm{\lblEFradius}; (C10) $\constref{chiG},\constref{chiY},\constref{chiR},\varepsilon$.}
\end{rulex}

\subsection{(b) Sensing plan: telemetry $\rightarrow$ automata triggers $\rightarrow$ coupling/MTC}\label{subsec:ops-sense}
\emph{\layertag{Stoch/Inf/Comp}}

\begin{definition}[Coupling mass]\label{def:coupling-mass}
With windowed, whitened mixed cumulants $\kappa_t$ and window Fisher $F_t$,
$(
M_t := \kappa_t^\top F_t^{-1}\kappa_t
)$
is the global coupling budget indicator.
\uses{(C5) $F$ and leakage $\chi$.}
\end{definition}

\begin{rulex}{Promotion gates (MTC over correlation)}\label{rule:promotion}
Promote only if \emph{all} hold: (i) held-out transport improvement at pre-registered MDE (power $\ge0.8$); (ii) SPRT on $M_t$ passes at $(\alpha,\beta)$ in the low-coupling regime; (iii) clamp in Green; (iv) if actuation uses IV, require $F>10$, pre-registered exclusion, and micro-ablation sign. Log the \texttt{sprt\_transcript\_id} and include \{H$_0$, H$_1$, $\delta$, $\alpha$, $\beta$\} in Tier-1. 
\uses{(C10) clamp; \jsgiiDef{\lblMTC}; \jsgiiDef{\lblIVbadge}.}
\end{rulex}

\begin{rulex}{DRO dual consistency (Δ)}\label{rule:dvdrho}
Before consuming any SPRT/DRO output, verify the envelope identity $d\,\mathbb E[\varphi_\eta]/d\rho\approx \eta^\star$ within $5\%$. If violated, \emph{quarantine} the window and freeze promotions. Emit a \texttt{dvdrho\_id} and add it to the window proof print.
\uses{\jsgiiThm{\lblDROenv}; \jsgiiDef{\lblQuarantine}.}
\end{rulex}

\begin{rulex}{Rights-plane tails \& safety trigger}\label{rule:rights-cvar}
Monitor the control barrier slack’s $\mathrm{CVaR}_{0.95}$. On breach, apply Freeze\,+\,Dwell escalation (Rule~\ref{rule:grazing}) and block promotions.
\uses{(C11) $\mathrm{CVaR}_{0.95}$; (C8) dwell escalation.}
\end{rulex}

\subsection{(c) Control plan: CLF/CBF-QP, PR/GKYP, switching stability, levers}\label{subsec:ops-ctrl}
\emph{\layertag{Ctrl/Rep/Alg}}

\begin{lemma}[Frontier-aware CBF relaxation]\label{lem:cbf-salt}
\emph{[\textup{Ctrl}]}
Off-tube enforce $L_f h_i + L_g h_i\,u \ge -\gamma_i h_i$. Inside tubes, a single-step audit with
\[
h_i^+\ \ge\ h_i^-\ -\ \sigma_i,\qquad \sigma_i:= L_{h_i}\,\constsymsalt\,\|\delta x\|
\]
controls saltation. If the residual lower bound $\mathrm{LB}:=h_i^+-h_i^-+\sigma_i$ is negative, reject the $\omega$-run. Produce a stable \texttt{cbf\_audit\_id} and include it in the proof print.
\uses{(C11) $\constsymsalt$, LB; (C1) tubes; Rule~\ref{rule:frontier-freeze}; \jsgiiLem{\lblSaltation}.}
\end{lemma}

\begin{proposition}[Band-limited PR with GKYP witness]\label{prop:gkyp-witness}
Certify $(\epsilon,[\constsymomegal,\constsymomegau])$-PR over a specified uncertainty set and emit $(\Omega,\epsilon,P,Q,\text{descriptor})$ plus a vertex hitmap and a stable \texttt{pr\_witness\_id}. If margins crowd $\constsymPRmargin$, shrink $\Omega$ automatically and re-certify. Record \texttt{pr\_witness\_id} in the proof print.
\uses{(C12) $\constsymomegal,\constsymomegau,\constsymPRmargin$.}
\end{proposition}

\begin{lemma}[Switching stability and degeneracy control]\label{lem:switching}
Block basis updates when the CI for $\Delta\sigma$ contains $0$. Otherwise either (i) furnish a CQLF certificate, or (ii) enforce a switch-dwell $\tau_{\min}^{\mathrm{sw}}$. If neither is feasible, \emph{k}-escalate to restore the spectral gap.
\uses{(C3) dwell; (C9) degeneracy policy; \jsgiiThm{\lblSwitchStab}.}
\end{lemma}

\begin{rulex}{Safe lever selection (normative dictionary)}\label{rule:levers}
Use safe-UCB with \emph{zero-violation} feasibility (QP\,+\,CVaR tails) before any lever pull. Reward is curvature-per-collateral (ISS improvement per unit fiscal/rights/holonomy cost). Record governance anchors (e.g., TRO/PI, FACA-like participation, OTA/GAO concurrence, anti-commandeering constraints).
\uses{(C11) $\mathrm{CVaR}$; \jsgiThm{\lblISS}.}
\end{rulex}

\subsection{(d) Proof-carrying certificates (transport, provenance, CIS, automata)}\label{subsec:ops-pcc}

\textbf{Tier-1 (Transport Pack; per window).} Hashes (FIM, whitening, features), BY table, $\kappa$, $M_t$, clamp state, \textit{EF guard id}, $\{\eta_k\}$ and dual gap, \texttt{kl\_quad\_bounds} verdict and $r$, CVaR slacks, $\constsymsalt$ with CI, \constref{Hol}/\constref{Harm}, PR witness id + hitmap id, solver transcript id, image/container SHA, risk posture, quarantine bit, \textit{SPRT params \{H$_0$,H$_1$, $\delta$, $\alpha$, $\beta$\} \& transcript id}, \textit{budget-guard verdict}, \textit{rollout mode}, and the \textit{window proof print (EF$\rightarrow$DRO$\rightarrow$PR$\rightarrow$CBF$\rightarrow\omega$)}.\\
\textbf{Tier-2 (Provenance ladder).} Hash-chain from telemetry $\to$ features $\to$ cumulants $\to$ FDR $\to$ MTC/IV $\to$ calibrations (saltation/DK) $\to$ QP/PR $\to$ clamp/quarantine $\to$ registries; enforce mirrored signed sources and parity checks ($\le5\%$ MAPE to $k\!=\!4$ moments, CVaR$_{0.95}$ preserved).\\
\textbf{Tier-3 (Coupling Identification Sheet).} One row per promoted index: BY thresholded $p$, MTC gates, achieved power, IV badge (F, exclusion contexts, ablation sign), preregistered dependence tests.\\
\textbf{Tier-4 ($\omega$-Automata).} Specs (e.g., $G(M_t\!\le\!B_{\mathrm{saf}})$, $G(h_i\!\ge\!0)$, $F(\hat R_t\!\ge\!R_{\min})$) with accept/reject and the exact KPI row-hashes consumed; replay must bit-match.
\uses{\jsgiiThm{\lblDROenv}; \jsgiiSec{\lblSchemas}.}
\end{definition}

\subsection{(e) KPIs, targets, horizon}\label{subsec:ops-kpis}
\emph{\layertag{Comp/Inf/Ctrl}}

\begin{mdframed}
\textbf{KPI policy (rolling $3\tau^\*$ windows; $\alpha=0.05$, $\beta=0.20$).}
\begin{enumerate}[leftmargin=*]
\item \textbf{Coupling mass} $M_t$: median $\downarrow$ by $\ge20\%$ over $2\tau^\*$; alarm if CI upper $>B_{\mathrm{saf}}$ or $\chi$ spikes. (I\textsubscript{3})
\item \textbf{Effective breadth} $|E_t|$: nonincreasing on stationary tasks; all promotions listed in CIS with MTC status. (I\textsubscript{3})
\item \textbf{Decision range} $\hat R_t=\prod_{i\le r}\sigma_i$ and $\log\hat R_t$: monotone $\uparrow$ subject to I\textsubscript{5}; alarm on conservativity residual. (I\textsubscript{5})
\item \textbf{Fisher drift \& EF radius} $r=\varepsilon/L_\psi$: stay within budget; two hits $\Rightarrow$ clamp escalate. (I\textsubscript{2}, I\textsubscript{3})
\item \textbf{Holonomy / Rotational Mass}: trend $\downarrow$; sustained rise $\Rightarrow$ k-escalation or switch-dwell. (I\textsubscript{4})
\item \textbf{Harmonic Rotational Mass}: flat or $\downarrow$; early degeneracy warning. (I\textsubscript{4})
\item \textbf{Frontier health} $(\constref{nuhat}, \constsymsalt, \text{duty}, \constref{pgraze})$: duty $\le10\%$; $\constref{nuhat}>0$. (I\textsubscript{1}, I\textsubscript{4})
\item \textbf{Chatter} (dwell violations): zero; any $\Rightarrow$ freeze + retune. (I\textsubscript{1})
\item \textbf{Rights-plane violations}: zero; any LB$<0$ on $h_i^+-h_i^-+\sigma_i$ $\Rightarrow$ run rejection. (I\textsubscript{5})
\item \textbf{Provenance coverage/latency}: $\ge99\%$, $\le1$ window; shortfall $\Rightarrow$ clamp + promotion freeze. (I\textsubscript{2})
\item \textbf{DRO dual-gap trend}: $\le 10^{-4}$; plateau $\Rightarrow$ solver/transport alarm. (I\textsubscript{3})
\item \textbf{Realized FDR}: $\le 0.9q$; exceedance $\Rightarrow$ deterministic tighten/freeze; relaxation permitted only after two Green windows within budget. (I\textsubscript{3})
\end{enumerate}
\end{mdframed}

\subsection*{Minimal rollout (five crisp steps)}\label{subsec:ops-rollout}
\emph{\layertag{Comp/Inf/Ctrl}}
\begin{enumerate}[leftmargin=*]
\item \textbf{Bring-up sensing:} window $\to$ FIM $\to$ whitening $\to$ cumulants; enforce \texttt{kl\_quad\_bounds} and start clamp FSM; emit Tier-1/2 ($\ge99\%$).
\item \textbf{Safe control core:} CLF/CBF-QP + frontier-aware relaxation + frozen $\Pi$; certify PR over uncertainty (GKYP); wire rights-plane CVaR.
\item \textbf{Identifiability gates:} enable MTC + IV badges; SPRT guarded by \texttt{dvdrho} check; log CIS \emph{and} add \texttt{sprt\_transcript\_id} to Tier-1.
\item \textbf{Dual-control on:} $\omega$-automata consume KPIs; monitors must accept before actuation; humans approve/decline; emit Tier-1…4 with replay pack and the window proof print.
\item \textbf{Self-scaling:} safe-UCB for levers; auto-retune controller band from PR hitmaps; tighten $(q,r)$ from realized FDR and provenance coverage.
\end{enumerate}

\subsection*{Stress tests (pre-expansion certification)}\label{subsec:ops-stress}
\emph{\layertag{Comp/Rep/Ctrl/Inf}}
\begin{itemize}[leftmargin=*]
\item \textbf{Non-smoothness:} induce controlled grazing to validate dwell boost \& saltation CI; reject if residual LB coverage $<$ nominal.
\item \textbf{Telemetry poisoning:} flip to signed-mirror sources; verify ladder parity; any divergence $\Rightarrow$ quarantine.
\item \textbf{Model drift:} force Red clamp; confirm fail-closed fallback (HSIC/CVaR-only) with zero rights violations.
\item \textbf{Identifiability:} simulate confounded instruments; ensure IV-badge exclusion test rejects with power.
\item \textbf{Controller degeneracy:} collapse gap; confirm k-escalation restores PR margin or switch-dwell enforces stability.
\end{itemize}

\begin{acceptchecks}
\item \textbf{Frontier Spend Ledger bound:}
print $C_0 N_{\text{cross}}$ and $(C_0/\tau_{\min})t$ using
\jsgiLem{\lblJacobiDefect} and \jsgiLem{\lblNoZeno}; pass if numerically consistent.
\item \textbf{Hol/HarmMass (DK/Wedin) plane-drift audit:}
for each crossing, verify $\angle(U^+,U^-)\le (1+\eta_{\mathrm{cal}})\,\text{const}/\delta_{\text{gap}}\ \|\Delta J\|_2$; pass if the replayed ledger matches stored SVDs within tolerance.
\item \textbf{EF guard (local KL–cumulant):}
evidence pack shows a \emph{passing} \texttt{kl\_quad\_bounds} with $(\varepsilon,\hat L_\psi,r)$ and \texttt{ef\_guard\_id} \emph{before} any Fisher-quadratic use, else fail-closed to HSIC/CVaR (\jsgiiThm{\lblEFradius}).
\item \textbf{SPRT precondition (DRO envelope):}
recompute finite-difference $dv/d\rho$ and compare to $\eta^\star$; pass if within 5\% (\jsgiiThm{\lblDROenv}); on fail, require quarantine bit and zero promotions (\jsgiiDef{\lblQuarantine}). Tier-1 carries \{H$_0$,H$_1$, $\delta$, $\alpha$, $\beta$\}.
\item \textbf{CBF saltation replay:}
rebuild $K$ from stored $(F^\pm,n)$; pass if $\|K-I\|_{\text{obs}}\le \overline C_{\text{salt}}(1+0.1)$ (\jsgiiLem{\lblSaltation}); else reject $\omega$-run. Proof print includes \texttt{cbf\_audit\_id}.
\item \textbf{Proof print order:} window proof print lists IDs in EF$\rightarrow$DRO$\rightarrow$PR$\rightarrow$CBF$\rightarrow\omega$ order; automata consume the same row-hash sequence.
\end{acceptchecks}
