% Robust constant key normalization: accepts either 'tau' or '\\tau'
% Requires xstring (loaded in jsg_common.tex)
% Simplified constkey: assume plain names like 'tau' (not '\\tau')
\providecommand{\constkey}[1]{\xdef\ConstKey{#1}}

% Fallback: if xstring's \IfBeginWith is unavailable for any reason, use a
% simplified constkey that assumes plain names (e.g., "tau" not "\tau").
% (No-op fallback retained intentionally)

% Constant reference helper: link to label const:<name> and print symbol constsym<name>
\renewcommand{\constref}[1]{\constkey{#1}\hyperref[const:\ConstKey]{\ensuremath{\csname constsym\ConstKey\endcsname}}}

% Clickable constant label (robust to '\\tau' vs 'tau')
\renewcommand{\constlabel}[1]{\constkey{#1}\label{const:\ConstKey}}

% Backward-compatibility alias for occasional typo
\providecommand{\constantlabel}[1]{\constlabel{#1}}
